class SentimentScore(BaseModel):
    """Brand sentiment analysis result"""
    score: float = Field(..., ge=0, le=100, description="Sentiment score (0=very negative, 50=neutral, 100=very positive)")
    trend: Literal["declining", "stable", "improving"] = Field(..., description="Sentiment trend over last 48 hours")
    confidence: float = Field(..., ge=0, le=100, description="Confidence in this assessment")
    
class BrandRiskAssessment(BaseModel):
    """Overall brand risk assessment"""
    current_sentiment_score: float = Field(..., ge=0, le=100)
    sentiment_trend: Literal["declining", "stable", "improving"]
    crisis_level: Literal["low", "medium", "high", "critical"]
    affected_channels: List[str] = Field(default_factory=list, description="Social media, customer support, news, etc.")
    competitor_threat: Literal["none", "low", "moderate", "high"]
    estimated_revenue_at_risk: Optional[float] = Field(None, description="Estimated revenue impact in USD")

class MarketingAction(BaseModel):
    """Single marketing action recommendation"""
    action_type: Literal[
        "pause_ads", 
        "launch_campaign", 
        "product_substitution", 
        "customer_notification",
        "social_media_response",
        "press_release",
        "loyalty_campaign",
        "retargeting",
        "content_creation"
    ]
    priority: Literal["immediate", "high", "medium", "low"]
    details: str = Field(..., description="Specific action details")
    timeline: str = Field(..., description="When to execute (e.g., 'within 2 hours', '24-48 hours')")
    budget_recommendation: Optional[float] = Field(None, description="Recommended budget in USD")
    expected_impact: str = Field(..., description="Expected outcome/benefit")
    rationale: str = Field(..., description="Why this action is recommended")

class AutoGeneratedContent(BaseModel):
    """Auto-generated marketing/communication content"""

    subject_line: Optional[str] = None
    body: str
    target_audience: str

@tool
def calculate_brand_damage_cost(
    sentiment_drop_points: float,
    affected_customers: int,
    average_customer_ltv: float,
    estimated_churn_rate: float = 0.15
) -> Dict[str, Any]:
    """
    Calculate estimated financial impact of brand damage.
    
    Args:
        sentiment_drop_points: How much sentiment score dropped (0-100 scale)
        affected_customers: Number of customers affected by the issue
        average_customer_ltv: Average customer lifetime value in USD
        estimated_churn_rate: Expected customer churn rate (default 15%)
    
    Returns:
        Financial impact analysis with revenue at risk and mitigation costs
    """
    
    # Estimate churn based on sentiment drop
    adjusted_churn_rate = min(estimated_churn_rate * (1 + sentiment_drop_points / 50), 0.50)
    
    # Calculate direct revenue impact
    estimated_churn_customers = int(affected_customers * adjusted_churn_rate)
    direct_revenue_loss = estimated_churn_customers * average_customer_ltv
    
    # Indirect costs
    acquisition_cost_to_replace = estimated_churn_customers * (average_customer_ltv * 0.25)  # Assume CAC is 25% of LTV
    word_of_mouth_impact = direct_revenue_loss * 0.30  # Negative WOM affects 30% more potential revenue
    
    total_brand_damage_cost = direct_revenue_loss + acquisition_cost_to_replace + word_of_mouth_impact
    
    # Mitigation budget recommendation
    recommended_mitigation_budget = total_brand_damage_cost * 0.15  # Spend 15% to save 85%
    
    return {
        "sentiment_drop_points": sentiment_drop_points,
        "affected_customers": affected_customers,
        "estimated_churn_rate": round(adjusted_churn_rate * 100, 2),
        "estimated_churn_customers": estimated_churn_customers,
        "direct_revenue_loss": round(direct_revenue_loss, 2),
        "acquisition_cost_to_replace": round(acquisition_cost_to_replace, 2),
        "word_of_mouth_impact": round(word_of_mouth_impact, 2),
        "total_brand_damage_cost": round(total_brand_damage_cost, 2),
        "recommended_mitigation_budget": round(recommended_mitigation_budget, 2),
        "roi_if_mitigated": round(((total_brand_damage_cost - recommended_mitigation_budget) / recommended_mitigation_budget) * 100, 2),
        "calculation_assumptions": {
            "average_customer_ltv": average_customer_ltv,
            "base_churn_rate": estimated_churn_rate,
            "cac_ltv_ratio": 0.25,
            "word_of_mouth_multiplier": 0.30,
            "recommended_mitigation_spend": 0.15
        },
        "calculated_at": datetime.now().isoformat()
    }
